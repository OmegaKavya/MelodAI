# Pinned deliberately, not loosely. This set has three mutually constraining
# edges and a floating version breaks at least one of them:
#
#   tensorflow 2.16 requires numpy<2 and protobuf<5
#   streamlit from 1.40 onward requires protobuf>=5.26
#   umap-learn 0.6 requires numpy>=2
#
# So streamlit and umap-learn are held below those boundaries. Loosening any
# line here produces an app that installs cleanly and fails at import.

# Deep learning
tensorflow==2.16.2
tensorflow-hub==0.16.1
protobuf<5
numpy<2

# Audio
librosa>=0.10,<1.0
soundfile>=0.12

# Representation and figures
opencv-python-headless>=4.8
matplotlib>=3.7,<4.0
scipy>=1.10

# Embedding map. The app places an uploaded song on the corpus map, which
# needs a projection that can transform unseen points; t-SNE cannot.
umap-learn==0.5.6
scikit-learn>=1.3,<2.0
joblib>=1.3

# Web app
streamlit==1.38.0

# tensorflow-hub imports pkg_resources, removed from setuptools 81.
setuptools<81

# Tests
pytest>=7.4
